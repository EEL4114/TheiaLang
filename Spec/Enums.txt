sum Unit
{
    enum<f64> Mass              // access via Unit.variants, here: Unit.variant = 0, so Unit.variants[0]
           {
               kg  = 1,         // member index 0, access via Mass.members;
               lbm = 0.453592,  // member index 1
           },
    enum<f64> Length            // Unit.variant = 1
            {
                m  = 1,         // member index 0
                ft = 0.3048,    // member index 1
                in = 0.0254,   // member index 2
            },
}

void OnUnitChanged(Unit newUnit, Unit oldUnit, @f64 associatedValue, string fieldName)
{
    if(newUnit == oldUnit)  // structural equality, not value equality: Unit.Length.m != Unit.Mass.kg
    {                       // values can be compared via newUnit.Value == oldUnit.value
        print($"{fieldName} already had unit {newUnit}");
        return;
    }

    associatedValue *= f64(newUnit) / f64(oldUnit); // alternatively newUnit.value / oldUnit.value
    return;
}

void OnUnitChangedV2(Unit newUnit, @f64 associatedValue, @Unit storedUnit, string fieldName)
{
    if(newUnit == storedUnit)
    {
        print($"{fieldName} already had unit {newUnit}");
        return;
    }

    if(newUnit.variant != storedUnit.variant)
    {
        print($"Dimension mismatch: {newUnit} is not compatible with {oldUnit}");
        return;
    }

    associatedValue *= newUnit.value / storedUnit.value;
    storedUnit = newUnit;

    return;
